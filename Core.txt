"""

Topological invariant calculator (Chern, Z₂) for 2D/3D systems

Production-grade implementation with guaranteed Hermiticity and unitarity

"""



import numpy as np

from dataclasses import dataclass, field

from typing import Dict, Optional, Tuple, Callable, TypeVar, Generic

import numpy.typing as npt

from functools import wraps

import warnings



# ============================================================

# 0. TYPE ANNOTATIONS AND UTILITIES

# ============================================================



T = TypeVar('T')

Vector = npt.NDArray[np.complex128]

Matrix = npt.NDArray[np.complex128]



@dataclass

class TopologicalResult:

    """Quantified topological invariant with error estimation."""

    raw: float

    quantized: int

    error: float

    method: str

    metadata: Dict = field(default_factory=dict)

    

    def __str__(self):

        return f"{self.method}: {self.quantized} (raw={self.raw:.6f}, error={self.error:.2e})"



def pauli_matrices() -> Tuple[Matrix, Matrix, Matrix, Matrix]:

    """Standard Pauli matrices."""

    σ0 = np.eye(2, dtype=np.complex128)

    σx = np.array([[0, 1], [1, 0]], dtype=np.complex128)

    σy = np.array([[0, -1j], [1j, 0]], dtype=np.complex128)

    σz = np.array([[1, 0], [0, -1]], dtype=np.complex128)

    return σ0, σx, σy, σz



def enforce_hermiticity(H: Matrix) -> Matrix:

    """Ensure matrix is Hermitian within numerical tolerance."""

    return (H + H.conj().T) / 2



def unitary_projection(M: Matrix, method: str = 'svd') -> Matrix:

    """

    Project onto U(N) using SVD or QR.

    

    Parameters:

    -----------

    M : Matrix

        Input matrix (m × n, m ≥ n)

    method : str

        'svd' (more stable) or 'qr' (faster)

    

    Returns:

    --------

    U : Matrix

        Closest unitary matrix to M

    """

    if method == 'svd':

        U, _, Vh = np.linalg.svd(M, full_matrices=False)

        return U @ Vh

    elif method == 'qr':

        Q, R = np.linalg.qr(M)

        # Remove phases from diagonal of R

        D = np.diag(np.sign(np.diag(R) + 1e-12))

        return Q @ D

    else:

        raise ValueError(f"Unknown projection method: {method}")

class Hamiltonian:

    """Base class for Bloch Hamiltonians."""

    

    def __init__(self, n_bands: int):

        self._n_bands = n_bands

    

    @property

    def num_bands(self) -> int:

        return self._n_bands

    

    def __call__(self, *k, **params) -> Matrix:

        raise NotImplementedError

    

    def eigen(self, *k, **params) -> Tuple[Vector, Matrix]:

        """Return eigenvalues and eigenvectors at k-point."""

        H = self(*k, **params)

        H = enforce_hermiticity(H)

        return np.linalg.eigh(H)

    

    def occupied_states(self, *k, n_occ: int, **params) -> Matrix:

        """Return occupied states matrix (n_bands × n_occ)."""

        _, vecs = self.eigen(*k, **params)

        return vecs[:, :n_occ]

# 2. MODELS (STRICTLY HERMITIAN)

# ============================================================



class QWZHamiltonian(Hamiltonian):

    """QWZ model: H = sin(kx)σx + sin(ky)σy + (m - cos(kx) - cos(ky))σz."""

    

    def __init__(self):

        super().__init__(2)

        self.σ0, self.σx, self.σy, self.σz = pauli_matrices()

    

    def __call__(self, kx: float, ky: float, m: float = 1.0) -> Matrix:

        dx = np.sin(kx)

        dy = np.sin(ky)

        dz = m - np.cos(kx) - np.cos(ky)

        H = dx * self.σx + dy * self.σy + dz * self.σz

        return enforce_hermiticity(H)



class HaldaneHamiltonian(Hamiltonian):

    """

    Haldane model on honeycomb lattice.

    Strictly Hermitian implementation with real dz component.

    """

    

    def __init__(self):

        super().__init__(2)

        self.σ0, self.σx, self.σy, self.σz = pauli_matrices()

    

    def __call__(self, kx: float, ky: float, 

                 t1: float = 1.0, t2: float = 0.2, 

                 phi: float = np.pi/2, m: float = 0.0) -> Matrix:

        # Nearest neighbors

        dx = t1 * (1 + np.cos(kx) + np.cos(ky))

        dy = t1 * (np.sin(kx) + np.sin(ky))

        

        # Next-nearest neighbors (real contributions only)

        # Using the standard form from Haldane's paper

        nn_vectors = [(1, 0), (0, 1), (-1, 1)]

        dz_term = 0.0

        for dx_vec, dy_vec in nn_vectors:

            k_dot = dx_vec * kx + dy_vec * ky

            dz_term += np.sin(k_dot + phi) + np.sin(k_dot - phi)

        

        dz = m - 2 * t2 * dz_term

        H = dx * self.σx + dy * self.σy + dz * self.σz

        return enforce_hermiticity(H)



class BHZHamiltonian(Hamiltonian):

    """Bernevig-Hughes-Zhang model (4×4) for quantum spin Hall effect."""

    

    def __init__(self):

        super().__init__(4)

        σ0, σx, σy, σz = pauli_matrices()

        

        # Dirac matrices in the BHZ basis

        self.Γ1 = np.kron(σx, σx)  # σx ⊗ σx

        self.Γ2 = np.kron(σx, σy)  # σx ⊗ σy

        self.Γ3 = np.kron(σx, σz)  # σx ⊗ σz

        self.Γ4 = np.kron(σy, σ0)  # σy ⊗ I

        self.Γ5 = np.kron(σz, σ0)  # σz ⊗ I

    

    def __call__(self, kx: float, ky: float, 

                 A: float = 1.0, B: float = 1.0, M: float = 0.0) -> Matrix:

        d0 = M - 2 * B * (2 - np.cos(kx) - np.cos(ky))

        d1 = A * np.sin(kx)

        d2 = A * np.sin(ky)

        

        H = d1 * self.Γ1 + d2 * self.Γ2 + d0 * self.Γ3

        return enforce_hermiticity(H)

# 3. DECORATORS FOR DISORDER AND DEFORMATIONS

# ============================================================



class DisorderedHamiltonian(Hamiltonian):

    """Adds Hermitian disorder without modifying base Hamiltonian."""

    

    def __init__(self, base: Hamiltonian, strength: float = 0.1, seed: Optional[int] = None):

        super().__init__(base.num_bands)

        self.base = base

        self.strength = strength

        

        if seed is not None:

            np.random.seed(seed)

    

    def __call__(self, *k, **params) -> Matrix:

        H0 = self.base(*k, **params)

        

        # Generate random Hermitian matrix

        n = self.num_bands

        disorder = np.random.randn(n, n) + 1j * np.random.randn(n, n)

        disorder = (disorder + disorder.conj().T) / np.sqrt(2)

        

        # Normalize and scale

        norm = np.linalg.norm(disorder, 'fro')

        if norm > 1e-12:

            disorder = disorder / norm * self.strength

        

        return H0 + disorder



class MomentumDeformedHamiltonian(Hamiltonian):

    """Applies momentum-space deformation k -> f(k)."""

    

    def __init__(self, base: Hamiltonian, 

                 deformation: Callable[[Tuple[float, ...]], Tuple[float, ...]]):

        super().__init__(base.num_bands)

        self.base = base

        self.deformation = deformation

    

    def __call__(self, *k, **params) -> Matrix:

        k_deformed = self.deformation(k)

        return self.base(*k_deformed, **params)

# 4. FHS CHERN CALCULATOR (2D)

# ============================================================



class FHSChernCalculator:

    """Fukui-Hatsugai-Suzuki method for Chern number in 2D."""

    

    def __init__(self, hamiltonian: Hamiltonian, n_occ: Optional[int] = None):

        self.H = hamiltonian

        self.n_occ = n_occ or (hamiltonian.num_bands // 2)

        

        if self.n_occ <= 0 or self.n_occ > hamiltonian.num_bands:

            raise ValueError(f"Invalid n_occ={n_occ} for {hamiltonian.num_bands} bands")

    

    def _compute_states(self, Nx: int, Ny: int, **params) -> Matrix:

        """Precompute occupied states on k-grid."""

        states = np.zeros((Nx, Ny, self.H.num_bands, self.n_occ), dtype=np.complex128)

        

        for i in range(Nx):

            for j in range(Ny):

                kx = 2 * np.pi * i / Nx

                ky = 2 * np.pi * j / Ny

                states[i, j] = self.H.occupied_states(kx, ky, n_occ=self.n_occ, **params)

        

        return states

    

    def compute(self, Nx: int = 30, Ny: int = 30, **params) -> TopologicalResult:

        """

        Compute Chern number using FHS method.

        

        Parameters:

        -----------

        Nx, Ny : int

            Number of k-points in each direction

        **params :

            Parameters passed to Hamiltonian

        

        Returns:

        --------

        TopologicalResult with Chern number

        """

        states = self._compute_states(Nx, Ny, **params)

        

        total_flux = 0.0

        for i in range(Nx):

            for j in range(Ny):

                # States at plaquette corners

                U00 = states[i, j]

                U10 = states[(i + 1) % Nx, j]

                U01 = states[i, (j + 1) % Ny]

                U11 = states[(i + 1) % Nx, (j + 1) % Ny]

                

                # Wilson links around plaquette

                L0 = unitary_projection(U00.conj().T @ U10)

                L1 = unitary_projection(U10.conj().T @ U11)

                L2 = unitary_projection(U11.conj().T @ U01)

                L3 = unitary_projection(U01.conj().T @ U00)

                

                # Berry curvature (determinant of plaquette)

                F = np.linalg.det(L0 @ L1 @ L2 @ L3)

                total_flux += np.angle(F)

        

        raw_chern = total_flux / (2 * np.pi)

        quantized = int(np.round(raw_chern))

        

        return TopologicalResult(

            raw=raw_chern,

            quantized=quantized,

            error=abs(raw_chern - quantized),

            method="FHS",

            metadata={'Nx': Nx, 'Ny': Ny, 'n_occ': self.n_occ}

        )

# 5. WILSON LOOP Z₂ CALCULATOR

# ============================================================



class WilsonLoopZ2Calculator:

    """

    Z₂ invariant via Wilson loop spectral flow (Fu-Kane-Mele).

    

    Implementation follows:

    - Fu, Kane, Mele (2007) for Z₂ in time-reversal invariant systems

    - Spectral flow of Wilson loop eigenvalues

    """

    

    def __init__(self, hamiltonian: Hamiltonian, n_occ: int):

        self.H = hamiltonian

        self.n_occ = n_occ

        

        if n_occ % 2 != 0:

            warnings.warn(f"n_occ={n_occ} is odd. Z₂ requires Kramers pairs.")

    

    def _wilson_loop_ky(self, ky: float, Nx: int, **params) -> Matrix:

        """Wilson loop W(ky) = Π_i U(ki, ky)^† U(ki+1, ky)."""

        W = np.eye(self.n_occ, dtype=np.complex128)

        

        for i in range(Nx):

            kx = 2 * np.pi * i / Nx

            kx_next = 2 * np.pi * (i + 1) / Nx

            

            # Get occupied states at adjacent k-points

            U_i = self.H.occupied_states(kx, ky, n_occ=self.n_occ, **params)

            U_next = self.H.occupied_states(kx_next, ky, n_occ=self.n_occ, **params)

            

            # Overlap matrix with unitary projection

            M = U_i.conj().T @ U_next

            M_unitary = unitary_projection(M, method='qr')  # Faster for Wilson loops

            

            W = W @ M_unitary

        

        return W

    

    def compute(self, Nx: int = 30, Ny: int = 30, **params) -> TopologicalResult:

        """

        Compute Z₂ invariant via Wilson loop spectral flow.

        

        The invariant is determined by counting parity of eigenvalue

        crossings at a reference line (θ=0) as ky goes from 0 to π.

        """

        # Compute Wilson loop eigenvalues for each ky

        eigenphases = []

        

        for j in range(Ny):

            ky = 2 * np.pi * j / Ny

            W = self._wilson_loop_ky(ky, Nx, **params)

            

            # Eigenvalues of Wilson loop: exp(iθ)

            eigvals = np.linalg.eigvals(W)

            phases = np.angle(eigvals)  # ∈ (-π, π]

            

            # Sort and store

            eigenphases.append(np.sort(phases))

        

        eigenphases = np.array(eigenphases)  # Shape: (Ny, n_occ)

        

        # Track spectral flow of the lowest band

        # For Z₂, we track flow of one Kramers pair

        flow = eigenphases[:, 0]  # Lowest eigenvalue

        

        # Unwrap phases to get continuous evolution

        flow_unwrapped = np.unwrap(flow)

        

        # Count crossings of π (mod 2π)

        # Z₂ = parity of number of crossings at θ=π

        # Equivalent to (θ(ky=π) - θ(ky=0)) / π mod 2

        delta_theta = flow_unwrapped[-1] - flow_unwrapped[0]

        z2_raw = delta_theta / np.pi

        

        # Quantize to 0 or 1

        z2_quantized = int(np.round(z2_raw)) % 2

        

        # Alternative method: count crossings of reference line

        crossings = 0

        for i in range(len(flow_unwrapped) - 1):

            # Check if line crosses π (mod 2π)

            if (flow_unwrapped[i] % (2*np.pi) < np.pi and 

                flow_unwrapped[i+1] % (2*np.pi) >= np.pi):

                crossings += 1

            elif (flow_unwrapped[i] % (2*np.pi) >= np.pi and 

                  flow_unwrapped[i+1] % (2*np.pi) < np.pi):

                crossings += 1

        

        z2_crossings = crossings % 2

        

        # Use the more reliable method

        z2_final = z2_quantized

        

        return TopologicalResult(

            raw=float(z2_raw),

            quantized=z2_final,

            error=abs(z2_raw - z2_final),

            method="Wilson-loop-Z2",

            metadata={

                'Nx': Nx, 

                'Ny': Ny, 

                'n_occ': self.n_occ,

                'crossings': crossings,

                'delta_theta': delta_theta

            }

        )

# 6. 3D SYSTEMS AND CHERN SLICES

# ============================================================



class Slice3DHamiltonian(Hamiltonian):

    """2D slice of a 3D Hamiltonian at fixed kz."""

    

    def __init__(self, h3d: Hamiltonian, kz: float):

        super().__init__(h3d.num_bands)

        self.h3d = h3d

        self.kz = kz

    

    def __call__(self, kx: float, ky: float, **params) -> Matrix:

        # Remove kz from params to avoid conflict

        params_no_kz = {k: v for k, v in params.items() if k != 'kz'}

        return self.h3d(kx, ky, self.kz, **params_no_kz)



class Chern3DScanner:

    """Scan Chern numbers across kz slices in 3D BZ."""

    

    def __init__(self, h3d: Hamiltonian, n_occ: Optional[int] = None):

        self.h3d = h3d

        self.n_occ = n_occ or (h3d.num_bands // 2)

    

    def scan_kz(self, Nkz: int = 20, Nxy: int = 20, **params) -> np.ndarray:

        """Compute Chern number for each kz slice."""

        chern_numbers = np.zeros(Nkz, dtype=int)

        

        for i in range(Nkz):

            kz = 2 * np.pi * i / Nkz

            h2d = Slice3DHamiltonian(self.h3d, kz)

            

            calculator = FHSChernCalculator(h2d, n_occ=self.n_occ)

            result = calculator.compute(Nx=Nxy, Ny=Nxy, **params)

            

            chern_numbers[i] = result.quantized

        

        return chern_numbers

def validate_hermiticity():

    """Validate that all models are strictly Hermitian."""

    print("=== Hermiticity Validation ===")

    

    models = [

        ("QWZ", QWZHamiltonian()),

        ("Haldane", HaldaneHamiltonian()),

        ("BHZ", BHZHamiltonian())

    ]

    

    for name, model in models:

        max_deviation = 0.0

        for _ in range(100):

            k = np.random.rand(model.num_bands) * 2 * np.pi

            H = model(*k[:2])

            deviation = np.max(np.abs(H - H.conj().T))

            max_deviation = max(max_deviation, deviation)

        

        status = "✓" if max_deviation < 1e-12 else "✗"

        print(f"{status} {name:10} max non-Hermiticity: {max_deviation:.2e}")



def benchmark_chern_qwz():

    """Benchmark Chern number calculation for QWZ model."""

    print(" === QWZ Chern Benchmark ===")

    

    model = QWZHamiltonian()

    

    # Known phase transitions at m = ±2, 0

    test_points = [

        (-2.5, 0, "Trivial"),

        (-1.5, -1, "Topological"),

        (0.5, 0, "Trivial"),

        (1.5, 1, "Topological"),

        (2.5, 0, "Trivial")

    ]

    

    for m, expected, desc in test_points:

        calculator = FHSChernCalculator(model)

        result = calculator.compute(Nx=40, Ny=40, m=m)

        

        correct = result.quantized == expected

        symbol = "✓" if correct else "✗"

        print(f"{symbol} m={m:5.2f}: C={result.quantized} "

              f"(expected {expected}), error={result.error:.2e} - {desc}")



def benchmark_z2_bhz():

    """Benchmark Z₂ calculation for BHZ model."""

    print(" === BHZ Z₂ Benchmark ===")

    

    model = BHZHamiltonian()

    

    # Known phase transitions at M = 0, ±4 (for B=1)

    test_points = [

        (-5.0, 0, "Trivial"),

        (-2.0, 1, "Topological"),

        (0.5, 1, "Topological"),

        (2.0, 1, "Topological"),

        (5.0, 0, "Trivial")

    ]

    

    for M, expected, desc in test_points:

        calculator = WilsonLoopZ2Calculator(model, n_occ=2)

        result = calculator.compute(Nx=30, Ny=30, M=M, A=1.0, B=1.0)

        

        correct = result.quantized == expected

        symbol = "✓" if correct else "✗"

        print(f"{symbol} M={M:5.2f}: Z₂={result.quantized} "

              f"(expected {expected}), error={result.error:.2e} - {desc}")



def test_disorder_robustness():

    """Test Chern number robustness under disorder."""

    print(" === Disorder Robustness Test ===")

    

    base_model = QWZHamiltonian()

    calculator = FHSChernCalculator(base_model)

    

    # Clean system

    clean_result = calculator.compute(Nx=30, Ny=30, m=1.5)

    print(f"Clean system: C={clean_result.quantized}")

    

    # With increasing disorder

    for strength in [0.1, 0.3, 0.5, 1.0]:

        disordered = DisorderedHamiltonian(base_model, strength=strength, seed=42)

        disordered_calc = FHSChernCalculator(disordered)

        result = disordered_calc.compute(Nx=30, Ny=30, m=1.5)

        

        print(f"Disorder {strength:.1f}: C={result.quantized} "

              f"(error={result.error:.2e})")



def demonstrate_3d_weyl():

    """Demonstrate 3D Weyl semimetal with varying Chern numbers."""

    print(" === 3D Weyl Semimetal ===")

    

    # Simple Weyl Hamiltonian

    class WeylHamiltonian(Hamiltonian):

        def __init__(self):

            super().__init__(2)

            self.σ0, self.σx, self.σy, self.σz = pauli_matrices()

        

        def __call__(self, kx: float, ky: float, kz: float, m: float = 2.0):

            dx = np.sin(kx)

            dy = np.sin(ky)

            dz = m - np.cos(kx) - np.cos(ky) - np.cos(kz)

            H = dx * self.σx + dy * self.σy + dz * self.σz

            return enforce_hermiticity(H)

    

    model = WeylHamiltonian()

    scanner = Chern3DScanner(model)

    

    chern_vs_kz = scanner.scan_kz(Nkz=40, Nxy=30, m=2.5)

    

    # Find Weyl nodes (where Chern number changes)

    weyl_positions = []

    for i in range(len(chern_vs_kz) - 1):

        if chern_vs_kz[i] != chern_vs_kz[i + 1]:

            kz_pos = 2 * np.pi * (i + 0.5) / len(chern_vs_kz)

            weyl_positions.append(kz_pos)

    

    print(f"Chern numbers vary from {np.min(chern_vs_kz)} to {np.max(chern_vs_kz)}")

    print(f"Found {len(weyl_positions)} Weyl node(s) at kz ≈ {weyl_positions}")

    

    return chern_vs_kz
